public bool run;


runsimulation(int replicationCount) {
run = true;
int currentReplication = 1;
beforeSimulation();
while(currentReplication < replicationCount && run == true) {
	beforeReplication();
	doReplication();
	afterReplication();
	currentReplication++;
}
afterSimulation();

}
